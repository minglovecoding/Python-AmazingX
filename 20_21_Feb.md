```c++
//Problem2-Modern Art 3
#include<iostream>
#include<cmath>
using namespace std;
const int Nmax=305;
int a[Nmax];
int dp[Nmax][Nmax];
int main(){
    int n;
    cin>>n;
    //initialize
    for(int i=0;i<n;i++){
        cin>>a[i];
        dp[i][i]=1;
    }
    //Interval DP
    for(int i=2;i<=n;i++){
        for(int l=0;l+i-1<n;l++){
            int r=l+i-1;
            //case 1
            dp[l][r]=1+dp[l+1][r];
            //case 2 
            for(int k=l+1;k<=r;k++){
                if(a[l]==a[k]){
                //1
                if(l+1==k){
                    dp[l][r]=min(dp[l][r],dp[k][r]);
                }//2
                else{
                    dp[l][r]=min(dp[l][r],dp[l+1][k-1]+dp[k][r]);
                }
                }
            }
        }
    }
    cout<<dp[0][n-1]<<"\n";
    return 0;
}
```

```c++
//Problem 1. Comfortable Cows
//1.每输入一头牛，如果这头牛位置之前没有牛，就添加这头牛计数cnt+1。
//2.每有一个新牛加入，就加入队列，并检查自己、上下左右四个点的状态。
//3.取出队首位置，如果上下左右有三个牛，在另一空白的位置加入牛
//4.每次加牛的时候，cnt++,加入队列并重复步骤3，可能会有新的连锁反应。
//5.队列为空时输出cnt-i
#include<bits/stdc++.h>
using namespace std;
int dx[4]={1,0,-1,0};
int dy[4]={0,1,0,-1};
set<pair<int,int>>cows;
queue<pair<int,int>>q;
int cnt=0;
bool hasCow(int x,int y){
    return cows.count({x,y})>0;
}

void addCow(int x,int y){
    if(hasCow(x,y)) return;
    cows.insert({x,y});
    cnt++;
    q.push({x,y});
}

void checkStatus(int x,int y){
    if(!hasCow(x,y)) return;
    int num=0;
    int emptyX=0;
    int emptyY=0;
    for(int i=0;i<4;i++){
        int nx=x+dx[i];
        int ny=y+dy[i];
        if(hasCow(nx,ny)) num++;
        else{
            emptyX=nx;
            emptyY=ny;
        }
    }
    if(num==3) addCow(emptyX,emptyY);
}

int main(){
    ios::sync_with_stdio(false);
    cin.tie(0);

    int N;
    cin>>N;

    for(int i=1;i<=N;i++){
        int x,y;
        cin>>x>>y;
        addCow(x,y);
        while(!q.empty()){
            auto [x,y]=q.front();
            q.pop();

            checkStatus(x,y);
            for(int i=0;i<4;i++){
                int nx=x+dx[i];
                int ny=y+dy[i];
                checkStatus(nx,ny);
            }
        }
        cout<<cnt-i<<"\n";
    }
    return 0;
}
```

```c++
//Problem 2. Year of the Cow
//1.以12年为一轮，统计哪些层有祖先，每一层距离就是12年。
//2.计算祖先层之间的层数距离，并加上第0层到最低层距离，并从大到小排序。
//3.第一次跳是从0层到最高层的距离
//4.以贪心方式选择前k-1个为时间穿越，剩余的累积12*层数差距
//5.输出累积值。
#include<bits/stdc++.h>
using namespace std;
using ll=long long;
int main(){
    ios::sync_with_stdio(false);
    cin.tie(0);
    
    int N,K;
    cin>>N>>K;

    vector<ll>layers;
    layers.push_back(0);
    for(int i=0;i<N;i++){
        ll year;
        cin>>year;
        ll layer=(year+11)/12;
        layers.push_back(layer);
    }
    
    //sort and unique
    sort(layers.begin(),layers.end());
    layers.erase(unique(layers.begin(),layers.end()),layers.end());

    ll answer=layers.back()*12;  //biggest gap time
    vector<ll>gaps;

    for(int i=1;i<layers.size();i++){
        ll empty=layers[i]-layers[i-1]-1; //must -1
        gaps.push_back(empty*12);
    }
    
    sort(gaps.rbegin(),gaps.rend());
    int skip=min(K-1,(int)gaps.size());
    
    for( int i=0;i<skip;i++){
        answer-=gaps[i];
    }
    
    cout<<answer<<"\n";
    return 0;
}
```

```c++
//Problem 3. Just Green Enough
//1.枚举矩形的左边界 L，再逐步向右移动右边界 R。
//2.R 每右移一列，就更新每一行在 [L,R] 内是否全部 >=100，以及是否含有 100。
//3.固定 [L,R] 后，从上到下扫描。小于 100 的行会截断连续合法区域。
//4.在连续合法区域中，记录起始行 start 和最近一个含 100 的行 last100。以当前行为下边界时，新增 last100 - start + 1 个矩形；尚未出现 100 则新增 0 个。
//5.累加答案。每个矩形只统计一次，时间复杂度 O(N^3)。
#include<bits/stdc++.h>
using namespace std;
const int Nmax=505;
int grid[Nmax][Nmax];
bool valid[Nmax];  //whether >=100
bool hundred[Nmax];  //is there has 100
int main(){
    ios::sync_with_stdio(false);
    cin.tie(0);
    int N;
    cin>>N;
    for(int i=0;i<N;i++){
        for(int j=0;j<N;j++){
            cin>>grid[i][j];
        }
    }
    long long sum=0;
    for(int l=0;l<N;l++){
        fill(valid,valid+N,true);
        fill(hundred,hundred+N,false);
        for(int r=l;r<N;r++){
            for(int row=0;row<N;row++){
                 valid[row]&=(grid[row][r]>=100);
                 hundred[row]|=(grid[row][r]==100);
            }
            int start=0;
            int end_100=-1;
            for(int row=0;row<N;row++){
                if(!valid[row]){
                    start=row+1;
                    end_100=-1;
                    continue;
                }
                if(hundred[row]){
                    end_100=row;
                }
                if(end_100!=-1){
                    sum+=end_100-start+1;
                }
            }
        }
    }
    cout<<sum<<endl;
    return 0;
}
```
